@user_router.message(CommandStart())
async def cmd_start(message: Message, command: CommandStart, state: FSMContext) -> None:
    await state.clear()
    user = message.from_user
    await db.add_user(user.id, user.full_name, user.username)

    is_admin = await db.is_admin(user.id)
    is_owner = (user.id == config.OWNER_ID)
    reply_kb = get_main_reply_keyboard(is_admin=is_admin, is_owner=is_owner)

    args = command.args
    if args:
        code = args.replace("anime_", "").strip()
        anime = await db.get_anime_by_code(code)
        if anime:
            await show_anime_details(message, anime, user.id)
            return

    await message.answer(
        f"👋 Assalomu alaykum, <b>{user.first_name}</b>!\n\n"
        f"🎬 <b>Eng sara va sifatli animelar botiga xush kelibsiz!</b>\n"
        f"Bu yerda siz o'zingiz sevgan animelarni o'zbek tilida yuqori sifatda tomosha qilishingiz mumkin.\n\n"
        f"<i>Qidirish uchun anime kodi yoki nomini yozing, yoki quyidagi tugmalardan foydalaning:</i>",
        reply_markup=reply_kb
    )


@user_router.callback_query(F.data == "check_sub")
async def callback_check_sub(callback: CallbackQuery, bot: Bot) -> None:
    user_id = callback.from_user.id
    channels = await db.get_channels()
    unsubscribed = []

    for ch in channels:
        try:
            m = await bot.get_chat_member(chat_id=ch["channel_id"], user_id=user_id)
            if m.status not in (
                ChatMemberStatus.MEMBER,
                ChatMemberStatus.ADMINISTRATOR,
                ChatMemberStatus.CREATOR
            ):
                unsubscribed.append(ch)
        except Exception:
            continue

    if unsubscribed:
        await callback.answer("❌ Hali barcha kanallarga a'zo bo'lmadingiz!", show_alert=True)
    else:
        await callback.answer("✅ Rahmat! Obuna tasdiqlandi.", show_alert=True)
        try:
            await callback.message.delete()
        except Exception:
            pass

        is_admin = await db.is_admin(user_id)
        is_owner = (user_id == config.OWNER_ID)
        await callback.message.answer(
            "🎉 <b>Tabriklaymiz! Siz muvaffaqiyatli obuna bo'ldingiz.</b>\n"
            "Endi botdan to'liq foydalanishingiz mumkin:",
            reply_markup=get_main_reply_keyboard(is_admin=is_admin, is_owner=is_owner)
        )

async def show_anime_details(target: Message | CallbackQuery, anime: dict, user_id: int) -> None:
    await db.increment_views(anime["id"])
    is_fav = await db.is_favorite(user_id, anime["id"])
    episodes = await db.get_episodes(anime["id"])
    has_episodes = len(episodes) > 0

    vip_badge = "👑 <b>[VIP ANIME]</b>\n" if anime.get("is_vip_only") else ""
    caption = (
        f"🎬 <b>{anime['title']}</b>\n\n"
        f"{vip_badge}"
        f"🔢 <b>Kodi:</b> <code>{anime['code']}</code>\n"
        f"🏷 <b>Janrlar:</b> {anime['genres']}\n"
        f"📅 <b>Yili:</b> {anime['year']}-yil\n"
        f"📌 <b>Holati:</b> {anime['status']}\n"
        f"👁 <b>Ko'rishlar:</b> {anime['views'] + 1}\n"
        f"🎞 <b>Mavjud qismlar:</b> {len(episodes)} ta\n\n"
        f"📖 <b>Tavsif:</b>\n<i>{anime['description']}</i>"
    )
    kb = get_anime_card_keyboard(anime["id"], anime["code"], is_fav, has_episodes)

    if isinstance(target, Message):
        if anime.get("poster_id"):
            try:
                await target.answer_photo(photo=anime["poster_id"], caption=caption, reply_markup=kb)
                return
            except Exception:
                pass
        await target.answer(caption, reply_markup=kb)
    else:
        if anime.get("poster_id"):
            try:
                await target.message.answer_photo(photo=anime["poster_id"], caption=caption, reply_markup=kb)
                await target.answer()
                return
            except Exception:
                pass
        await target.message.answer(caption, reply_markup=kb)
        await target.answer()

@user_router.message(SearchFSM.waiting_query)
async def process_search_query(message: Message, state: FSMContext) -> None:
    query = message.text.strip()
    user_id = message.from_user.id
    is_admin = await db.is_admin(user_id)
    is_owner = (user_id == config.OWNER_ID)

    anime_by_code = await db.get_anime_by_code(query)
    if anime_by_code:
        await state.clear()
        await message.answer("✅ Anime topildi!", reply_markup=get_main_reply_keyboard(is_admin, is_owner))
        await show_anime_details(message, anime_by_code, user_id)
        return

    results = await db.search_anime_by_title(query)
    await state.clear()

    if not results:
        await message.answer(
            f"❌ Kechirasiz, <b>'{query}'</b> bo'yicha hech qanday anime topilmadi.\n"
            f"Kodni yoki nomni to'g'ri yozganingizga ishonch hosil qiling.",
            reply_markup=get_main_reply_keyboard(is_admin, is_owner)
        )
        return

    if len(results) == 1:
        await message.answer("✅ Anime topildi!", reply_markup=get_main_reply_keyboard(is_admin, is_owner))
        await show_anime_details(message, results[0], user_id)
        return

    text = f"🔎 <b>'{query}' bo'yicha topilgan animelar:</b>\n\n"
    for idx, a in enumerate(results, start=1):
        text += f"{idx}. <b>{a['title']}</b> | Kodi: <code>{a['code']}</code> ({a['year']}-yil)\n"
    text += "\n<i>Tomosha qilish uchun anime kodini botga yuboring!</i>"
    await message.answer(text, reply_markup=get_main_reply_keyboard(is_admin, is_owner))

@user_router.callback_query(F.data.startswith("fav_toggle_"))

    anime_id = int(callback.data.replace("fav_toggle_", ""))
    user_id = callback.from_user.id
    is_now_fav = await db.toggle_favorite(user_id, anime_id)
    anime = await db.get_anime_by_id(anime_id)
    if not anime:
        await callback.answer("Anime topilmadi.")
        return

    episodes = await db.get_episodes(anime_id)
    new_kb = get_anime_card_keyboard(anime_id, anime["code"], is_now_fav, len(episodes) > 0)
    try:

        pass

    if is_now_fav:
        await callback.answer("⭐ Sevimlilar ro'yxatiga qo'shildi!", show_alert=False)
    else:
        await callback.answer("❌ Sevimlilardan olib tashlandi.", show_alert=False)


@user_router.callback_query(F.data.startswith("episodes_"))
async def callback_show_episodes(callback: CallbackQuery) -> None:
    parts = callback.data.split("_")
    anime_id = int(parts[1])
    page = int(parts[2]) if len(parts) > 2 else 1

    anime = await db.get_anime_by_id(anime_id)
    if not anime:
        await callback.answer("Anime topilmadi.")
        return

    if anime.get("is_vip_only"):
        is_user_vip = await db.is_vip(callback.from_user.id)
        if not is_user_vip and callback.from_user.id != config.OWNER_ID:
            await callback.answer(
                "👑 Bu anime faqat VIP foydalanuvchilar uchun ochiq!",
                show_alert=True
            )
            return

    episodes = await db.get_episodes(anime_id)
    if not episodes:
        await callback.answer("Ushbu anime uchun hali qismlar yuklanmagan.", show_alert=True)
        return

    kb = get_episodes_keyboard(anime_id, episodes, page=page)
    text = (
        f"🎬 <b>{anime['title']}</b>\n"
        f"Kerakli qismni tanlang:"
    )
    try:
        await callback.message.edit_text(text, reply_markup=kb)
    except Exception:
        await callback.message.answer(text, reply_markup=kb)
    await callback.answer()

@user_router.callback_query(F.data.startswith("quality_"))
async def callback_send_video(callback: CallbackQuery, bot: Bot) -> None:
    parts = callback.data.split("_")
    episode_id = int(parts[1])
    quality = parts[2]

    ep = await db.get_episode_by_id(episode_id)
    if not ep:
        await callback.answer("Qism topilmadi.", show_alert=True)
        return

    anime = await db.get_anime_by_id(ep["anime_id"])
    file_id = None
    if quality == "480":
        file_id = ep.get("file_id_480")
    elif quality == "720":
        file_id = ep.get("file_id_720")
    elif quality == "1080":
        file_id = ep.get("file_id_1080")
    else:
        file_id = ep.get("file_id_720") or ep.get("file_id_480") or ep.get("file_id_1080")

    if not file_id:
        await callback.answer("Bu sifatdagi video hali yuklanmagan!", show_alert=True)
        return

    caption = (
        f"🎬 <b>{anime['title']}</b> | <b>{ep['episode_number']}-qism</b>\n"
        f"📺 Sifat: {quality}p\n\n"
        f"🤖 Bot: @{(await bot.get_me()).username}"
    )
    await callback.answer("Video yuborilmoqda...")
    try:
        await callback.message.answer_video(video=file_id, caption=caption)
    except Exception as e:
        await callback.message.answer(f"❌ Videoni yuborishda xatolik yuz berdi: {e}")
